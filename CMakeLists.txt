project(mulacover_cpp VERSION 0.1.0 LANGUAGES C CXX)

    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
endif()
option(MULACOVER_CUDA "Build the shared GGML CUDA backend" OFF)
option(MULACOVER_VULKAN "Build the shared GGML Vulkan backend" OFF)
option(MULACOVER_METAL "Build the shared GGML Metal backend" OFF)
option(MULACOVER_STATIC "Build a static generator library" ON)
option(MULACOVER_BUILD_SERVER "Build the native HTTP server and embedded UI" ON)
option(MULACOVER_USE_INSTALLED_GGML "Explicitly use an installed GGML package instead of the HeartCodec source pin" OFF)
set(MULACOVER_HEARTCODEC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../heartcodec.cpp" CACHE PATH "Shared HeartCodec checkout")
if(DEFINED MULACOVER_GGML_SOURCE_DIR AND NOT "${MULACOVER_GGML_SOURCE_DIR}" STREQUAL "")
    message(FATAL_ERROR "MULACOVER_GGML_SOURCE_DIR was retired after GGML #22. Use HeartCodec's coordinated submodule or a parent-provided GGML target in a fresh build directory.")
endif()
# An installed GGML package lets Spark reuse its existing CUDA build.
if(MULACOVER_USE_INSTALLED_GGML AND NOT TARGET ggml AND NOT TARGET ggml::ggml)
    find_package(ggml CONFIG REQUIRED)
endif()
if(NOT TARGET heartcodec::heartcodec)
    if(EXISTS "${MULACOVER_HEARTCODEC_DIR}/CMakeLists.txt")
        set(HEARTCODEC_BUILD_TOOLS OFF CACHE BOOL "")
        set(HEARTCODEC_BUILD_TESTS OFF CACHE BOOL "")
        set(HEARTCODEC_CUDA ${MULACOVER_CUDA} CACHE BOOL "" FORCE)
        set(HEARTCODEC_VULKAN ${MULACOVER_VULKAN} CACHE BOOL "" FORCE)
        set(HEARTCODEC_METAL ${MULACOVER_METAL} CACHE BOOL "" FORCE)
        add_subdirectory("${MULACOVER_HEARTCODEC_DIR}" heartcodec)
    else()
        find_package(heartcodec CONFIG REQUIRED)
    endif()
endif()
if(TARGET ggml)
    set(mulacover_ggml ggml)
else()
    set(mulacover_ggml ggml::ggml)
endif()
set(mulacover_sources src/model.cpp src/graph.cpp src/session.cpp src/sampling.cpp src/api.cpp src/runtime.cpp src/continuation.cpp src/job.cpp
    src/conditioning.cpp src/text.cpp src/score.cpp src/style.cpp)

    add_library(mulacover STATIC ${mulacover_sources})

    add_library(mulacover SHARED ${mulacover_sources})
    target_compile_definitions(mulacover PUBLIC MULACOVER_SHARED PRIVATE MULACOVER_BUILD)

add_library(mulacover::mulacover ALIAS mulacover)
target_include_directories(mulacover PUBLIC include PRIVATE src)

target_link_libraries(mulacover PRIVATE heartcodec::heartcodec ${mulacover_ggml})

if(MULACOVER_BUILD_SERVER)
    find_package(Threads REQUIRED)
    find_package(Git QUIET)
    set(MULACOVER_SOURCE_COMMIT "unknown")
    set(MULACOVER_SOURCE_DIRTY 1)
    if(GIT_FOUND)
        execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            OUTPUT_VARIABLE MULACOVER_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE git_status)
        if(NOT git_status EQUAL 0)
            set(MULACOVER_SOURCE_COMMIT "unknown")
        endif()
        execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            OUTPUT_VARIABLE source_changes OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE git_status)
        if(git_status EQUAL 0 AND source_changes STREQUAL "")
            set(MULACOVER_SOURCE_DIRTY 0)
        endif()
    endif()
    set(MULACOVER_DEFAULT_BACKEND "cpu")
    foreach(backend METAL VULKAN CUDA)
        if(GGML_${backend})
            string(TOLOWER ${backend} MULACOVER_DEFAULT_BACKEND)
        endif()
    endforeach()
    add_library(mulacover-server-support STATIC tools/server/http.cpp tools/server/json.cpp tools/server/multipart.cpp tools/server/base64.cpp tools/server/service.cpp tools/server/transcription.cpp)
    target_include_directories(mulacover-server-support PUBLIC tools)
    target_link_libraries(mulacover-server-support PUBLIC mulacover Threads::Threads)
    target_compile_definitions(mulacover-server-support PUBLIC MULACOVER_VERSION="${PROJECT_VERSION}"
        MULACOVER_SOURCE_COMMIT="${MULACOVER_SOURCE_COMMIT}" MULACOVER_SOURCE_DIRTY=${MULACOVER_SOURCE_DIRTY}
        MULACOVER_DEFAULT_BACKEND="${MULACOVER_DEFAULT_BACKEND}")
    if(WIN32)
        target_link_libraries(mulacover-server-support PUBLIC ws2_32)
    endif()
    set(MULACOVER_UI_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/mulacover_ui_html.h")
    add_custom_command(OUTPUT "${MULACOVER_UI_HEADER}"
        COMMAND ${CMAKE_COMMAND} -DINPUT=${CMAKE_CURRENT_SOURCE_DIR}/tools/ui/index.html -DOUTPUT=${MULACOVER_UI_HEADER}
            -DSYMBOL=mulacover_ui_html -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_resource.cmake
        DEPENDS tools/ui/index.html cmake/embed_resource.cmake VERBATIM)
    add_executable(mulacover-server tools/mulacover-server.cpp "${MULACOVER_UI_HEADER}")
    target_include_directories(mulacover-server PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(mulacover-server PRIVATE mulacover-server-support)
endif()
