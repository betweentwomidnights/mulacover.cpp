cmake_minimum_required(VERSION 3.20)
project(mulacover_cpp VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
option(MULACOVER_CUDA "Build the shared GGML CUDA backend" OFF)
option(MULACOVER_STATIC "Build a static generator library" ON)
option(MULACOVER_USE_INSTALLED_GGML "Explicitly use an installed GGML package instead of the HeartCodec source pin" OFF)
set(MULACOVER_HEARTCODEC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../heartcodec.cpp" CACHE PATH "Shared HeartCodec checkout")
set(MULACOVER_GGML_SOURCE_DIR "" CACHE PATH "Isolated GGML development checkout; does not change shared pins")
if(MULACOVER_GGML_SOURCE_DIR)
    if(MULACOVER_USE_INSTALLED_GGML OR TARGET ggml OR TARGET ggml::ggml)
        message(FATAL_ERROR "The development GGML source must supply the single GGML instance")
    endif()
    find_package(Git REQUIRED)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/ggml-development.json" development_profile)
    string(JSON development_revision GET "${development_profile}" revision)
    execute_process(COMMAND ${GIT_EXECUTABLE} -C "${MULACOVER_GGML_SOURCE_DIR}" rev-parse HEAD
        OUTPUT_VARIABLE source_revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE source_status)
    execute_process(COMMAND ${GIT_EXECUTABLE} -C "${MULACOVER_GGML_SOURCE_DIR}" status --porcelain
        OUTPUT_VARIABLE source_dirty OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE dirty_status)
    if(NOT source_status EQUAL 0 OR NOT dirty_status EQUAL 0 OR source_dirty OR NOT source_revision STREQUAL development_revision)
        message(FATAL_ERROR "Development GGML must be clean at ${development_revision}")
    endif()
    message(STATUS "MuLaCover-only GGML development revision: ${source_revision}; shared consumer pins unchanged")
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
    set(GGML_BUILD_TESTS OFF CACHE BOOL "")
    set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "")
    set(GGML_CUDA ${MULACOVER_CUDA} CACHE BOOL "" FORCE)
    add_subdirectory("${MULACOVER_GGML_SOURCE_DIR}" ggml-development)
endif()
# An installed GGML package lets Spark reuse its existing CUDA build.
if(MULACOVER_USE_INSTALLED_GGML AND NOT TARGET ggml AND NOT TARGET ggml::ggml)
    find_package(ggml CONFIG REQUIRED)
endif()
if(NOT TARGET heartcodec::heartcodec)
    if(EXISTS "${MULACOVER_HEARTCODEC_DIR}/CMakeLists.txt")
        set(HEARTCODEC_BUILD_TOOLS OFF CACHE BOOL "")
        set(HEARTCODEC_BUILD_TESTS OFF CACHE BOOL "")
        set(HEARTCODEC_CUDA ${MULACOVER_CUDA} CACHE BOOL "" FORCE)
        add_subdirectory("${MULACOVER_HEARTCODEC_DIR}" heartcodec)
    else()
        find_package(heartcodec CONFIG REQUIRED)
    endif()
endif()
if(TARGET ggml)
    set(mulacover_ggml ggml)
else()
    set(mulacover_ggml ggml::ggml)
endif()
set(mulacover_sources src/model.cpp src/graph.cpp src/session.cpp src/sampling.cpp src/api.cpp src/continuation.cpp src/job.cpp)
if(MULACOVER_STATIC)
    add_library(mulacover STATIC ${mulacover_sources})
else()
    add_library(mulacover SHARED ${mulacover_sources})
    target_compile_definitions(mulacover PUBLIC MULACOVER_SHARED PRIVATE MULACOVER_BUILD)
endif()
add_library(mulacover::mulacover ALIAS mulacover)
target_include_directories(mulacover PUBLIC include PRIVATE src)
target_link_libraries(mulacover PRIVATE heartcodec::heartcodec ${mulacover_ggml})
set_target_properties(mulacover PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
    target_compile_options(mulacover PRIVATE -ffp-contract=off)
endif()
add_executable(mulacover-parity tools/parity.cpp)
target_link_libraries(mulacover-parity PRIVATE mulacover ${mulacover_ggml})
add_executable(mulacover-generate tools/generate.cpp)
add_executable(mulacover-continue tools/continue.cpp)
target_link_libraries(mulacover-continue PRIVATE mulacover)
target_include_directories(mulacover-generate PRIVATE src)
target_link_libraries(mulacover-generate PRIVATE mulacover heartcodec::heartcodec ${mulacover_ggml})
add_executable(mulacover-quantize tools/quantize.cpp)
target_include_directories(mulacover-quantize PRIVATE src)
target_link_libraries(mulacover-quantize PRIVATE ${mulacover_ggml})
add_executable(mulacover-sampling-check tools/sampling-check.cpp src/sampling.cpp)
target_include_directories(mulacover-sampling-check PRIVATE src include)
if(BUILD_TESTING)
    add_executable(test-job tests/job.cpp)
    target_link_libraries(test-job PRIVATE mulacover)
    add_test(NAME job-api COMMAND test-job)
    add_executable(test-continuation tests/continuation.cpp)
    target_include_directories(test-continuation PRIVATE src)
    target_link_libraries(test-continuation PRIVATE mulacover)
    add_test(NAME continuation COMMAND test-continuation)
    add_executable(test-tempo tests/tempo.cpp)
    target_include_directories(test-tempo PRIVATE src include)
    add_test(NAME tempo COMMAND test-tempo)
    add_executable(test-sampling tests/sampling.cpp src/sampling.cpp)
    target_include_directories(test-sampling PRIVATE src include)
    add_test(NAME sampling COMMAND test-sampling)
    add_executable(test-mulacover-api tests/api.c)
    target_link_libraries(test-mulacover-api PRIVATE mulacover)
    add_test(NAME mulacover-api COMMAND test-mulacover-api)
endif()
